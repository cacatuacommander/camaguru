src/
  server.js          // creates http server, delegates to router
  router.js          // matches method+path to a handler
  routes/
    auth.js           // POST /api/register, /api/login, /api/logout, /api/confirm/:token, /api/reset-password
    users.js          // GET/PATCH /api/users/me
    images.js         // GET /api/images (paginated), POST /api/images, DELETE /api/images/:id
    gallery.js         // GET /api/gallery?page=
    comments.js        // POST /api/images/:id/comments
    likes.js            // POST/DELETE /api/images/:id/like
  controllers/         // one file per route group, the actual logic
  models/               // raw SQL query functions per table (users, images, comments, likes)
  middleware/
    session.js           // parses cookie, attaches req.user
    bodyParser.js         // reads/parses JSON or multipart bodies manually
  lib/
    db.js                 // pg Pool setup
    mailer.js              // SMTP sending
    hash.js                  // wraps crypto/bcrypt
public/                       // static HTML/CSS/JS served as-is
